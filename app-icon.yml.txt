# Updates the home-screen icon to match the month / holiday (and the icons picked in Settings).
# Runs every day shortly after midnight Eastern time; you can also run it by hand from the Actions tab.
name: Update app icon
on:
  schedule:
    - cron: "17 5 * * *"
  workflow_dispatch:
permissions:
  contents: write
  pages: write
jobs:
  icon:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
      - run: npm install --no-save @resvg/resvg-js@2.6.2
      - run: node tools/app-icon.mjs
        env:
          APP_URL: ${{ secrets.APP_URL }}
          APP_KEY: ${{ secrets.APP_KEY }}
      - name: Save changes
        run: |
          git config user.name "app-icon bot"
          git config user.email "app-icon-bot@users.noreply.github.com"
          git add -A appicon manifest.webmanifest
          if git diff --cached --quiet; then echo "No icon change today."; exit 0; fi
          git commit -m "Update app icon"
          git push
          gh api -X POST "repos/${GITHUB_REPOSITORY}/pages/builds" || true
        env:
          GH_TOKEN: ${{ github.token }}
